# Пояснения по лабораторной работе №14

Эта работа продолжает цикл работ по одномерным массивам: теперь действие над массивом зависит от того, выполняется ли условие для всего массива.

Как и в лабораторной работе №13, массив можно ввести вручную или заполнить случайными числами. Элементы в этих заданиях целые (для удобства деления они хранятся в массиве типа `float`), а нумерация элементов начинается с нуля – так же, как в выводе программы.

## Что нужно прочитать перед началом:
1. Основы работы с массивами
2. Циклы `for` и условный оператор `if-else`
3. Функции `fmod()` (остаток от деления вещественных чисел) и `fabs()` (модуль) из библиотеки `math.h`

Итак, когда все условия понятны, разберём код решения:
```c
#include <stdio.h> // Подключение библиотеки для ввода/вывода
#include <stdlib.h> // Подключение библиотеки для работы с функциями генерации случайных чисел
#include <time.h> // Подключение библиотеки для работы со временем
#include <math.h> // Подключение библиотеки для математических функций (остаток fmod, модуль fabs)
#include <windows.h> // Подключение модуля windows.h для установки кодировки вывода

int main() { // Объявляем главную функцию
    SetConsoleOutputCP(CP_UTF8); // Устанавливаем кодировку UTF-8 для вывода русских символов
    int n, choice; // Объявляем переменные для размера массива и выбора способа ввода

    printf("Введите размер массива A: "); // Запрашиваем размер массива
    scanf("%d", &n); // Считываем размер массива

    float A[n]; // Объявляем массив A из n элементов

    printf("Выберите способ ввода массива A (1 - ручной, 2 - автоматический): "); // Запрашиваем способ ввода массива
    scanf("%d", &choice); // Считываем выбор способа ввода

    if (choice == 1) { // Если выбран ручной ввод...
        printf("Введите элементы массива A:\n"); // ...просим ввести элементы массива
        for (int i = 0; i < n; i++) { // Перебираем все элементы массива
            printf("A[%d] = ", i); // Выводим приглашение для ввода очередного элемента
            scanf("%f", &A[i]); // Считываем элемент массива
        }
    } else if (choice == 2) { // Иначе, если выбран автоматический ввод...
        srand(time(NULL)); // Инициализируем генератор случайных чисел текущим временем
        printf("Сгенерированный массив A:\n"); // ...сообщаем, что выводим сгенерированный массив
        for (int i = 0; i < n; i++) { // Перебираем все элементы массива
            A[i] = rand() % 131 - 30; // Заполняем элемент случайным целым числом от -30 до 100
            printf("A[%d] = %.2f\n", i, A[i]); // Выводим сгенерированный элемент
        }
    } else { // Иначе...
        printf("Некорректный выбор способа ввода.\n"); // ...сообщаем об ошибке
        return 1; // Завершаем программу с кодом ошибки 1
    }

    int noMultiple = 1; // Предполагаем, что элементов, кратных трём, нет
    for (int i = 0; i < n; i++) { // Перебираем все элементы массива
        if (fmod(A[i], 3) == 0) { // Если такой элемент нашёлся...
            noMultiple = 0; // ...запоминаем, что предположение неверно
        }
    }

    if (noMultiple) { // Проверяем условие: в массиве нет элементов, кратных трём
        printf("\nУсловие выполнено.\n"); // Сообщаем, что условие выполнено
        for (int i = 0; i < n; i++) { // Перебираем все элементы массива
            if (fmod(A[i], 2) != 0) { // Если элемент нечётный...
                A[i] = A[i] / 2; // ...изменяем его
            }
        }
        printf("\nМассив после преобразования:\n"); // Выводим заголовок результата
        for (int i = 0; i < n; i++) { // Перебираем все элементы массива
            printf("A[%d] = %.2f\n", i, A[i]); // Выводим элемент массива
        }
    } else { // Иначе...
        printf("\nУсловие не выполнено.\n"); // Сообщаем, что условие не выполнено
        int index = -1; // Номер элемента (максимальный из меньших 50); -1 – такого элемента ещё не нашли
        for (int i = 0; i < n; i++) { // Перебираем все элементы массива
            if (A[i] < 50) { // Если элемент подходит...
                if (index == -1 || A[i] > A[index]) { // Если это первый подходящий или он больше текущего...
                    index = i; // ...запоминаем его номер
                }
            }
        }
        if (index == -1) { // Если подходящего элемента не нашлось...
            printf("\nТаких элементов нет.\n"); // Сообщаем, что подходящих элементов нет
        } else { // Иначе...
            printf("\nМаксимальный элемент, меньший 50: %.2f\n", A[index]); // Выводим найденное значение
        }
    }

    return 0; // Завершаем программу с возвращением 0
}
```

> ```
